/* ------------------------------------------------------- ATS CALCULATION FAVORITE WINS AN EXACT-SPREAD RESULT ------------------------------------------------------- */ function pickCovered(game, pickedTeam) { const homeScore = Number(game.home_score); const awayScore = Number(game.away_score); if ( Number.isNaN(homeScore) || Number.isNaN(awayScore) || game.home_score === null || game.away_score === null ) { return false; } const homeTeam = game.home_team; const pickedScore = pickedTeam === homeTeam ? homeScore : awayScore; const opponentScore = pickedTeam === homeTeam ? awayScore : homeScore; const spread = parseSpreadForTeam( game.spread, pickedTeam ); if (spread === null) { return false; } const adjustedScore = pickedScore + spread; // Clear cover: picked team wins. if (adjustedScore > opponentScore) { return true; } // Clear loss: picked team loses. if (adjustedScore < opponentScore) { return false; } // Exact spread: favorite is the sole winner. return spread < 0; }
